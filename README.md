<!-- EXECUTION PIPELINE -->
<table width="100%">
  <tr>
    <td align="center" bgcolor="#0d1117">
      <h2>🤖 Execution Pipeline & Decision Logic</h2>
    </td>
  </tr>
  <tr>
    <td>
      <br />
      <ul>
        <li><b>1. Initialization:</b> Bootstraps Chrome via Selenium WebDriver and navigates directly to the Cookie Clicker web app.</li>
        <li><b>2. High-Frequency Loop:</b> Sends non-stop, low-latency click events to the main cookie element.</li>
        <li><b>3. Periodic Evaluation:</b> Scans active DOM selectors every 5 seconds checking for <code>.upgrade.enabled</code> and <code>.product.enabled</code> classes.</li>
        <li><b>4. Dynamic Optimization:</b> Prioritizes top-tier upgrades and buildings to compound exponential CPS progression.</li>
      </ul>
      <br />
    </td>
  </tr>
</table>
